🏗️ Architecture Overview
Technology Stack

Framework: Vendure (NestJS-based e-commerce framework)
Database: SQLite (dev) / PostgreSQL (production)
API: GraphQL
Authentication: JWT + Session-based
Real-time: WebSocket subscriptions
File Storage: Local filesystem / S3 compatible

Server URLs

Shop API: http://localhost:3000/shop-api - Customer-facing API
Admin API: http://localhost:3000/admin-api - Admin operations
Admin UI: http://localhost:3002/admin - Web-based admin panel
Assets: http://localhost:3000/assets - Product images
Dev Mailbox: http://localhost:3000/mailbox - Email testing

JavaScript/TypeScript Client Example
typescript// shop-api-client.ts
import { ApolloClient, InMemoryCache, gql } from '@apollo/client';

const client = new ApolloClient({
  uri: 'http://localhost:3000/shop-api',
  cache: new InMemoryCache(),
  credentials: 'include', // Include cookies
});

// Get products
async function getProducts() {
  const { data } = await client.query({
    query: gql`
      query GetProducts {
        products(options: { take: 10 }) {
          items {
            id
            name
            slug
            variants {
              id
              price
              customFields {
                color
                voltage
              }
            }
          }
          totalItems
        }
      }
    `
  });
  return data.products;
}

// Add to cart
async function addToCart(variantId: string, quantity: number) {
  const { data } = await client.mutate({
    mutation: gql`
      mutation AddToCart($variantId: ID!, $quantity: Int!) {
        addItemToOrder(productVariantId: $variantId, quantity: $quantity) {
          ... on Order {
            id
            total
            lines {
              productVariant {
                name
              }
              quantity
            }
          }
          ... on InsufficientStockError {
            errorCode
            message
            quantityAvailable
          }
        }
      }
    `,
    variables: {
      variantId,
      quantity
    }
  });
  return data.addItemToOrder;
}

// B2B: Get bulk pricing
async function getBulkPricing(variantId: string, quantity: number) {
  const { data } = await client.query({
    query: gql`
      query GetBulkPrice($variantId: ID!, $quantity: Int!) {
        calculateBulkPrice(productVariantId: $variantId, quantity: $quantity) {
          unitPrice
          totalPrice
          discountPercent
          discountAmount
        }
      }
    `,
    variables: {
      variantId,
      quantity
    }
  });
  return data.calculateBulkPrice;
}

// Usage
async function main() {
  // Get products
  const products = await getProducts();
  console.log('Products:', products);

  // Add first variant to cart
  if (products.items.length > 0 && products.items[0].variants.length > 0) {
    const variantId = products.items[0].variants[0].id;
    const cartResult = await addToCart(variantId, 2);
    console.log('Cart:', cartResult);

    // Check bulk pricing
    const bulkPrice = await getBulkPricing(variantId, 100);
    console.log('Bulk price for 100 units:', bulkPrice);
  }
}

main().catch(console.error);
React Integration Example
tsx// ProductList.tsx
import React, { useEffect, useState } from 'react';
import { useQuery, useMutation } from '@apollo/client';
import { GET_PRODUCTS, ADD_TO_CART } from './queries';

function ProductList() {
  const { loading, error, data } = useQuery(GET_PRODUCTS, {
    variables: { take: 20 }
  });
  
  const [addToCart] = useMutation(ADD_TO_CART);
  const [notification, setNotification] = useState('');

  const handleAddToCart = async (variantId: string) => {
    try {
      const result = await addToCart({
        variables: { variantId, quantity: 1 }
      });
      
      if (result.data.addItemToOrder.__typename === 'Order') {
        setNotification('Added to cart!');
      } else {
        setNotification('Error: ' + result.data.addItemToOrder.message);
      }
    } catch (err) {
      setNotification('Error adding to cart');
    }
  };

  if (loading) return <p>Loading...</p>;
  if (error) return <p>Error: {error.message}</p>;

  return (
    <div>
      {notification && <div className="notification">{notification}</div>}
      
      <div className="product-grid">
        {data.products.items.map((product: any) => (
          <div key={product.id} className="product-card">
            <img src={product.featuredAsset?.preview} alt={product.name} />
            <h3>{product.name}</h3>
            <p>{product.description}</p>
            
            {/* Custom fields for electronics */}
            {product.customFields.manufacturer && (
              <p>Brand: {product.customFields.manufacturer}</p>
            )}
            {product.customFields.warrantyInfo && (
              <p>Warranty: {product.customFields.warrantyInfo}</p>
            )}
            
            {product.variants.map((variant: any) => (
              <div key={variant.id} className="variant">
                <span>{variant.name}</span>
                <span>${(variant.price / 100).toFixed(2)}</span>
                
                {/* Variant custom fields */}
                {variant.customFields.color && (
                  <span>Color: {variant.customFields.color}</span>
                )}
                {variant.customFields.cableLength && (
                  <span>Length: {variant.customFields.cableLength}m</span>
                )}
                
                <button onClick={() => handleAddToCart(variant.id)}>
                  Add to Cart
                </button>
              </div>
            ))}
          </div>
        ))}
      </div>
    </div>
  );
}
